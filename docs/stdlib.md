// Encoder writes one JSON value, usually an object or an array, adding the commas, colons and
// quotes itself. Object and array open a container and end closes the innermost one; inside an
// object, key names the member whose value comes next. Values are string, int, float, bool, null
// and raw (JSON encoded already):
//
// 	e := json.new_encoder(buf[:0])
// 	e.object()
// 	e.key("id")
// 	e.int(7)
// 	e.key("tags")
// 	e.array()
// 	e.string("a")
// 	e.end()
// 	e.end()
// 	if e.finish() == nil {
// 		w.write(e.bytes()) // {"id":7,"tags":["a"]}
// 	}
//
// Nothing is written that would make the output invalid: a call out of place (a value in an object
// without a key, key outside an object or twice, end with nothing open or right after a key, a
// second top-level value) sets err to Misuse, nesting deeper than max_depth sets Depth, a NaN
// or infinite float sets Float, and raw of invalid JSON sets Syntax. The first error is kept,
// and every call after it writes nothing. Finish also checks that the value is complete.
//
// The output is appended to the slice given to new_encoder. When it is full, a call sets Full
// and writes nothing at all (each call writes all of its output or none); with new_encoder_in the
// slice grows into the arena instead. To stream through a small buffer, write e.bytes() out and
// call e.clear() whenever it fills: Clear empties the buffer but keeps the nesting and comma state,
// so the next call continues the same document. A call that fails with Full can be retried
// after draining: set e.err = nil first.
//
// The output is compact unless set_indent is called first: then it has a member or element per
// line, indented by nesting.
struct Encoder {
	err error
	// fields that aren't pub
}

impl Encoder {
	// bytes returns the output written so far (since the last clear).
	fn bytes(&self) -> u8[] in self

	// clear empties the output buffer but keeps the encoder's state (the open containers, whether a
	// comma comes next, err), so the calls after it continue the same document: write e.bytes()
	// somewhere, then call e.clear(), to stream a large document through a small buffer.
	fn clear(&self)

	// set_indent lays out the rest of the output: each member or element of a non-empty object or array starts a new line holding
	// prefix and then indent once per level of nesting, the closing bracket starts a line one level
	// out, a colon is followed by a space, and empty objects and arrays stay {} and []. The first line
	// has no prefix and the last no line break. Raw values are laid out the same way. Call it before the first value: afterwards it sets
	// Misuse.
	fn set_indent(&self, prefix str in self, indent str in self)

	// finish returns err, after setting it to Misuse if the document isn't complete: nothing was
	// written, or an array or object is still open.
	fn finish(&self) -> !

	// object starts an object: Key and a value for each member, then end.
	fn object(&self)

	// array starts an array: its elements, then end.
	fn array(&self)

	// end closes the innermost open object or array.
	fn end(&self)

	// key writes the name of an object's next member; its value comes next.
	fn key(&self, name str)

	// string writes s as a JSON string, escaped exactly as append_string does.
	fn string(&self, s str)

	// int writes an integer.
	fn int(&self, v int)

	// float writes the shortest decimal that reads back as v (strconv.append_float with prec -1:
	// exponent notation below 1e-6 and from 1e21, as JavaScript writes numbers). JSON has no NaN or
	// infinity: those set Float.
	fn float(&self, v f64)

	// bool writes true or false.
	fn bool(&self, v bool)

	// null writes null.
	fn null(&self)

	// raw writes b, one complete JSON value encoded already (from another Encoder, a database, a
	// file), as the next value. It is checked as Decoder.skip checks a value: invalid JSON sets
	// Syntax and writes nothing. It is written as it is, whitespace and all, unless set_indent was
	// called: then its whitespace is replaced by the layout's.
	fn raw(&self, b u8[])

	// raw_start is raw for a value the caller writes itself, for one too large to go through the
	// buffer: it writes what comes before a value here (a comma, a line break and indentation, or

// Decoder pulls values out of a JSON document. The first error is kept in err, and every call
// after it fails.
struct Decoder {
	err error
	// fields that aren't pub
}

impl Decoder {
	// begin_object consumes '{'.
	fn begin_object(&self) -> bool

	// begin_array consumes '['.
	fn begin_array(&self) -> bool

	// more reports whether the current array has another element (and consumes the separator).
	fn more(&self) -> bool

	// field reads the next object key into keybuf, returning false at the end of the object.
	fn field(&self, keybuf u8[]) -> (str, bool) in keybuf

	// key reads the next object key, as field does, into the decoder's arena (new_decoder_in) at its
	// exact size (see new_decoder_in for how long it may be kept). NoArena on a decoder from
	// new_decoder, Buffer if the arena is full.
	fn key(&self) -> (str, bool) in self

	// Buffer if the decoded string doesn't fit in buf.

	// size (see new_decoder_in for how long it may be kept). NoArena on a decoder from new_decoder,
	// Buffer if the arena is full; the decoded string is measured first, so a full arena uses none
	// of it. text_max also bounds the length.
	fn text(&self) -> (str, bool) in self

	// text_max is text for a string of at most max bytes, decoded (as UTF-8, invalid bytes counting as
	// the 3 of U+FFFD): a longer one sets TooLong and uses none of the arena, so one string in
	// untrusted input can't fill it. After TooLong or Buffer the decoder is at the string's
	// opening quote (Line and offset point at it).
	fn text_max(&self, max int) -> (str, bool) in self

	// int decodes an integer. A number with a fraction or exponent is Type (it is consumed).
	fn int(&self) -> (int, bool)

	// float decodes a number as the nearest float64 (an integer too). Too large for a float64 is
	// Type.
	fn float(&self) -> (f64, bool)

	// bool decodes true or false.
	fn bool(&self) -> (bool, bool)

	// null consumes a null if it is the next value and reports whether it was (a missing or optional
	// value; anything else is left for the other methods).
	fn null(&self) -> bool

	// skip skips one value of any type, checking that it is valid JSON: brackets match, commas and
	// colons are where they belong, escapes are valid and strings hold no raw control characters
	// (invalid UTF-8 is accepted, as by string). It works iteratively; arrays and objects may nest up
	// to max_depth deep.
	fn skip(&self) -> bool

	// offset returns how many bytes of the input have been consumed (at most len(data)).
	fn offset(&self) -> int

	// line returns the 1-based line the decoder has reached (where an error was found, after one).
	fn line(&self) -> int

	// end reports whether only whitespace remains.
	fn end(&self) -> bool
}

// wildcard: {name} matches one segment, {name...} (last) matches the rest of the path, and {$}

// one that matches none gets 404 Not Found. Registering a malformed or duplicate pattern, or more
// than max_routes, panics.
